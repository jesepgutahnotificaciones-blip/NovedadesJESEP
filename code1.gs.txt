    return SpreadsheetApp.getActiveSpreadsheet();

  /* =====================================================
    6. FECHA / FORMATO
    ===================================================== */

  function fechaActual_() {

    return new Date();

  }


  function formatearFecha_(fecha) {

    if (!fecha) return '';

    try {

      return Utilities.formatDate(
        new Date(fecha),
        Session.getScriptTimeZone(),
        'dd/MM/yyyy'
      );

    } catch (e) {

      return fecha;

    }

  }


  /* =====================================================
    7. NORMALIZACIÓN
    ===================================================== */

  function normalizarTexto_(valor) {

    if (valor === null || valor === undefined) {
      return '';
    }

    return String(valor)
      .trim()
      .toUpperCase();

  }


  function normalizarCC_(valor) {

    if (valor === null || valor === undefined) {
      return '';
    }

    return String(valor)
      .replace(/[.\- ]/g, '')
      .trim();

  }


  /* =====================================================
    8. HASH DE CONTRASEÑA
    ===================================================== */

  function hashPassword_(password) {

    password = String(password || '');

    const digest = Utilities.computeDigest(
      Utilities.DigestAlgorithm.SHA_256,
      password,
      Utilities.Charset.UTF_8
    );

    return digest
      .map(function(byte) {

        const v = byte < 0 ? byte + 256 : byte;

        return ('0' + v.toString(16)).slice(-2);

      })
      .join('');

  }


  /* =====================================================
    9. TOKEN DE SESIÓN
    ===================================================== */

  function crearToken_(usuario) {

    const token = Utilities.getUuid();

    const cache = CacheService.getScriptCache();

    cache.put(
      'SESION_' + token,
      JSON.stringify(usuario),
      CACHE_SEGUNDOS
    );

    return token;

  }


  /* =====================================================
    10. VALIDAR SESIÓN
    ===================================================== */

  function validarSesion_(token) {

    if (!token) {

      throw new Error('Sesión no válida o expirada.');

    }

    const cache = CacheService.getScriptCache();

    const datos = cache.get('SESION_' + token);

    if (!datos) {

      throw new Error(
        'La sesión ha expirado. Inicie sesión nuevamente.'
      );

    }

    try {

      return JSON.parse(datos);

    } catch (e) {

      throw new Error('Sesión inválida.');

    }

  }


  /* =====================================================
    11. CERRAR SESIÓN
    ===================================================== */

  function cerrarSesion(token) {

    if (!token) {
      return {
        estado: true
      };
    }

    CacheService
      .getScriptCache()
      .remove('SESION_' + token);

    return {
      estado: true,
      mensaje: 'Sesión cerrada correctamente.'
    };

  }


  /* =====================================================
    12. OBTENER USUARIO
    ===================================================== */

  function obtenerUsuario_(usuario) {

    const ss = getSS_();

    const hoja = ss.getSheetByName(HOJA_USUARIOS);

    if (!hoja) {

      throw new Error(
        'No existe la hoja USUARIOS.'
      );

    }

    const valores = hoja.getDataRange().getValues();

    if (valores.length < 2) {
      return null;
    }

    const encabezados = valores[0];

    const idxUsuario =
      encabezados.indexOf('USUARIO');

    const idxHash =
      encabezados.indexOf('PASSWORD_HASH');

    const idxRol =
      encabezados.indexOf('ROL');

    const idxDependencia =
      encabezados.indexOf('DEPENDENCIA');

    const idxEstado =
      encabezados.indexOf('ESTADO');

    if (
      idxUsuario === -1 ||
      idxHash === -1 ||
      idxRol === -1 ||
      idxDependencia === -1 ||
      idxEstado === -1
    ) {

      throw new Error(
        'La hoja USUARIOS no tiene la estructura requerida.'
      );

    }

    usuario = normalizarTexto_(usuario);

    for (let i = 1; i < valores.length; i++) {

      if (
        normalizarTexto_(valores[i][idxUsuario]) === usuario
      ) {

        return {
          fila: i + 1,
          usuario: valores[i][idxUsuario],
          passwordHash: valores[i][idxHash],
          rol: valores[i][idxRol],
          dependencia: valores[i][idxDependencia],
          estado: valores[i][idxEstado]
        };

      }

    }

    return null;

  }


  /* =====================================================
    13. VALIDAR USUARIO
    ===================================================== */

  function validarUsuario(usuario, clave) {

    usuario = normalizarTexto_(usuario);

    if (!usuario || !clave) {

      return {
        estado: false,
        mensaje: 'Debe ingresar usuario y contraseña.'
      };

    }

    const datos = obtenerUsuario_(usuario);

    if (!datos) {

      registrarAuditoria_(
        usuario,
        'LOGIN_FALLIDO',
        '',
        'Usuario inexistente'
      );

      return {
        estado: false,
        mensaje: 'Usuario o contraseña incorrectos.'
      };

    }

    if (
      normalizarTexto_(datos.estado) !== 'ACTIVO'
    ) {

      registrarAuditoria_(
        usuario,
        'LOGIN_RECHAZADO',
        '',
        'Usuario inactivo'
      );

      return {
        estado: false,
        mensaje: 'El usuario se encuentra inactivo.'
      };

    }

    const hashIngresado =
      hashPassword_(clave);

    if (hashIngresado !== datos.passwordHash) {

      registrarAuditoria_(
        usuario,
        'LOGIN_FALLIDO',
        '',
        'Contraseña incorrecta'
      );

      return {
        estado: false,
        mensaje: 'Usuario o contraseña incorrectos.'
      };

    }

    actualizarUltimoAcceso_(usuario);

    const sesion = {

      usuario: datos.usuario,

      rol: normalizarTexto_(datos.rol),

      dependencia:
        normalizarTexto_(datos.dependencia)

    };

    const token = crearToken_(sesion);

    registrarAuditoria_(
      datos.usuario,
      'LOGIN_EXITOSO',
      '',
      'Ingreso al sistema'
    );

    return {

      estado: true,

      token: token,

      usuario: datos.usuario,

      rol: sesion.rol,

      dependencia: sesion.dependencia

    };

  }


  /* =====================================================
    14. ACTUALIZAR ÚLTIMO ACCESO
    ===================================================== */

  function actualizarUltimoAcceso_(usuario) {

    const ss = getSS_();

    const hoja = ss.getSheetByName(HOJA_USUARIOS);

    if (!hoja) return;

    const valores = hoja.getDataRange().getValues();

    if (valores.length < 2) return;

    const encabezados = valores[0];

    const idxUsuario =
      encabezados.indexOf('USUARIO');

    const idxAcceso =
      encabezados.indexOf('ULTIMO_ACCESO');

    if (
      idxUsuario === -1 ||
      idxAcceso === -1
    ) {
      return;
    }

    for (let i = 1; i < valores.length; i++) {

      if (
        normalizarTexto_(valores[i][idxUsuario]) ===
        normalizarTexto_(usuario)
      ) {

        hoja
          .getRange(i + 1, idxAcceso + 1)
          .setValue(new Date());

        break;

      }

    }

  }


  /* =====================================================
    15. OBTENER DATOS DE SESIÓN
    ===================================================== */

  function obtenerSesion(token) {

    const sesion =
      validarSesion_(token);

    return {

      estado: true,

      usuario: sesion.usuario,

      rol: sesion.rol,

      dependencia: sesion.dependencia

    };

  }


  /* =====================================================
    16. VALIDACIÓN DE DEPENDENCIA
    ===================================================== */

  function validarDependencia_(sesion, dependencia) {

    if (
      !sesion ||
      !sesion.dependencia
    ) {

      return false;

    }

    const rol =
      normalizarTexto_(sesion.rol);

    if (
      rol === 'ADMINISTRADOR'
    ) {

      return true;

    }

    const depUsuario =
      normalizarTexto_(sesion.dependencia);

    const depDato =
      normalizarTexto_(dependencia);

    return (
      depUsuario === depDato
    );

  }


  /* =====================================================
    17. OBTENER ENCABEZADOS
    ===================================================== */

  function obtenerEncabezados_(hoja) {

    if (!hoja) {
      throw new Error('Hoja no encontrada.');
    }

    const ultimaColumna =
      hoja.getLastColumn();

    if (ultimaColumna === 0) {
      return [];
    }

    return hoja
      .getRange(1, 1, 1, ultimaColumna)
      .getValues()[0]
      .map(function(x) {

        return String(x)
          .trim();

      });

  }


  /* =====================================================
    18. CREAR ESTRUCTURA NOVEDADES
    ===================================================== */

  function inicializarHojaNovedades() {

    const ss = getSS_();

    let hoja =
      ss.getSheetByName(HOJA_NOVEDADES);

    if (!hoja) {

      hoja =
        ss.insertSheet(HOJA_NOVEDADES);

    }

    const encabezados =
      COLUMNAS_NOVEDADES.slice();

    hoja
      .getRange(
        1,
        1,
        1,
        encabezados.length
      )
      .setValues([encabezados]);

    hoja
      .getRange(1, 1, 1, encabezados.length)
      .setFontWeight('bold');

    hoja.setFrozenRows(1);

    return {
      estado: true,
      mensaje:
        'Hoja NOVEDADES configurada correctamente.'
    };

  }


  /* =====================================================
    19. VALIDAR ESTRUCTURA NOVEDADES
    ===================================================== */

  function validarEstructuraNovedades_() {

    const ss = getSS_();

    const hoja =
      ss.getSheetByName(HOJA_NOVEDADES);

    if (!hoja) {

      throw new Error(
        'No existe la hoja NOVEDADES.'
      );

    }

    const encabezados =
      obtenerEncabezados_(hoja);

    for (
      let i = 0;
      i < COLUMNAS_NOVEDADES.length;
      i++
    ) {

      if (
        encabezados[i] !==
        COLUMNAS_NOVEDADES[i]
      ) {

        throw new Error(
          'La estructura de NOVEDADES no coincide. ' +
          'La columna ' +
          (i + 1) +
          ' debe ser "' +
          COLUMNAS_NOVEDADES[i] +
          '".'
        );

      }

    }

    return hoja;

  }


  /* =====================================================
    20. OBTENER FUNCIONARIO POR CC
    ===================================================== */

  function obtenerFuncionarioPorCedula_(cedula) {

    const ss = getSS_();

    const hoja =
      ss.getSheetByName(HOJA_BASE);

    if (!hoja) {

      throw new Error(
        'No existe la hoja LISTADO_BASE.'
      );

    }

    const datos =
      hoja.getDataRange().getValues();

    if (datos.length < 2) {
      return null;
    }

    const encabezados =
      datos[0].map(function(x) {
        return String(x).trim();
      });

    const idxCC =
      encabezados.indexOf('CEDULA');

    if (idxCC === -1) {

      throw new Error(
        'No existe la columna CEDULA en LISTADO_BASE.'
      );

    }

    const ccBuscada =
      normalizarCC_(cedula);

    for (let i = 1; i < datos.length; i++) {

      if (
        normalizarCC_(datos[i][idxCC]) ===
        ccBuscada
      ) {

        return construirFuncionario_(
          encabezados,
          datos[i],
          i + 1
        );

      }

    }

    return null;

  }


  /* =====================================================
    21. CONSTRUIR OBJETO FUNCIONARIO
    ===================================================== */

  function construirFuncionario_(
    encabezados,
    fila,
    numeroFila
  ) {

    const obj = {

      fila: numeroFila,

      datos: {}

    };

    encabezados.forEach(function(encabezado, i) {

      obj.datos[encabezado] =
        fila[i];

    });

    obj.cedula =
      obj.datos['CEDULA'];

    obj.nivel =
      obj.datos['NIV'];

    obj.grado =
      obj.datos['GR'];

    obj.funcionario =
      obj.datos['FUNCIONARIO'];

    obj.dependencia =
      obj.datos['DEPENDENCIA'];

    obj.turno =
      obj.datos['TURNO'];

    obj.placaChip =
      obj.datos['Placa_Chip'];

    return obj;

  }


  /* =====================================================
    22. BUSCAR FUNCIONARIO
    ===================================================== */

  function buscarFuncionario(token, valor) {

    const sesion =
      validarSesion_(token);

    valor =
      String(valor || '').trim();

    if (!valor) {

      return {
        estado: false,
        mensaje:
          'Ingrese una cédula o nombre para realizar la búsqueda.'
      };

    }

    const ss = getSS_();

    const hoja =
      ss.getSheetByName(HOJA_BASE);

    if (!hoja) {

      throw new Error(
        'No existe la hoja LISTADO_BASE.'
      );

    }

    const datos =
      hoja.getDataRange().getValues();

    if (datos.length < 2) {

      return {
        estado: true,
        resultados: []
      };

    }

    const encabezados =
      datos[0].map(function(x) {

        return String(x).trim();

      });

    const idxCC =
      encabezados.indexOf('CEDULA');

    const idxNombre =
      encabezados.indexOf('FUNCIONARIO');

    const idxDep =
      encabezados.indexOf('DEPENDENCIA');

    if (
      idxCC === -1 ||
      idxNombre === -1
    ) {

      throw new Error(
        'LISTADO_BASE debe contener CEDULA y FUNCIONARIO.'
      );

    }

    const consulta =
      normalizarTexto_(valor);

    const consultaCC =
      normalizarCC_(valor);

    const resultados = [];

    for (let i = 1; i < datos.length; i++) {

      const cc =
        normalizarCC_(datos[i][idxCC]);

      const nombre =
        normalizarTexto_(datos[i][idxNombre]);

      const coincideCC =
        consultaCC &&
        cc.indexOf(consultaCC) !== -1;

      const coincideNombre =
        nombre.indexOf(consulta) !== -1;

      if (
        coincideCC ||
        coincideNombre
      ) {

        const dependencia =
          idxDep !== -1
            ? datos[i][idxDep]
            : '';

        if (
          !validarDependencia_(
            sesion,
            dependencia
          )
        ) {

          continue;

        }

        resultados.push(
          construirFuncionario_(
            encabezados,
            datos[i],
            i + 1
          )
        );

      }

    }

    registrarAuditoria_(
      sesion.usuario,
      'BUSQUEDA_FUNCIONARIO',
      valor,
      'Resultados: ' + resultados.length
    );

    return {

      estado: true,

      resultados: resultados

    };

  }


  /* =====================================================
    23. OBTENER TODAS LAS NOVEDADES DE UN FUNCIONARIO
    ===================================================== */

  function obtenerHistorialCC_(cedula) {

    const hoja =
      validarEstructuraNovedades_();

    const ultimaFila =
      hoja.getLastRow();

    if (ultimaFila < 2) {
      return [];
    }

    const datos =
      hoja
        .getRange(
          2,
          1,
          ultimaFila - 1,
          COLUMNAS_NOVEDADES.length
        )
        .getValues();

    const idxCC = 2;

    const ccBuscada =
      normalizarCC_(cedula);

    const historial = [];

    datos.forEach(function(fila, indice) {

      if (
        normalizarCC_(fila[idxCC]) ===
        ccBuscada
      ) {

        const obj = {};

        COLUMNAS_NOVEDADES.forEach(
          function(col, i) {

            obj[col] = fila[i];

          }
        );

        obj.fila =
          indice + 2;

        historial.push(obj);

      }

    });

    return historial;

  }


  /* =====================================================
    24. CONSULTAR HISTORIAL
    ===================================================== */

  function consultarHistorial(token, cedula) {

    const sesion =
      validarSesion_(token);

    const funcionario =
      obtenerFuncionarioPorCedula_(cedula);

    if (!funcionario) {

      return {

        estado: false,

        mensaje:
          'No se encontró el funcionario en LISTADO_BASE.'

      };

    }

    if (
      !validarDependencia_(
        sesion,
        funcionario.dependencia
      )
    ) {

      throw new Error(
        'No tiene autorización para consultar esta dependencia.'
      );

    }

    const historial =
      obtenerHistorialCC_(cedula);

    registrarAuditoria_(
      sesion.usuario,
      'CONSULTA_HISTORIAL',
      cedula,
      'Registros encontrados: ' + historial.length
    );

    return {

      estado: true,

      funcionario: funcionario,

      historial: historial

    };

  }


  /* =====================================================
    25. GENERAR TEXTO DE NOVEDAD
    ===================================================== */

  function generarTextoNovedad_(datos) {

    const novedad =
      normalizarTexto_(datos.NOVEDAD);

    const descripcion =
      String(datos.DESCRIPCION || '').trim();

    const dias =
      String(datos.Dias || '').trim();

    const fechaInicial =
      datos['Fecha INICIAL']
        ? formatearFecha_(datos['Fecha INICIAL'])
        : '';

    const fechaPresentacion =
      datos['Fecha PRESENTACION']
        ? formatearFecha_(datos['Fecha PRESENTACION'])
        : '';

    const observacion =
      String(datos.Observacion || '').trim();

    let texto = novedad;

    if (descripcion) {

      texto +=
        ' - ' +
        descripcion;

    }

    if (dias) {

      texto +=
        ' - ' +
        dias +
        ' DÍA(S)';

    }

    if (fechaInicial) {

      texto +=
        ' - FECHA INICIAL: ' +
        fechaInicial;

    }

    if (fechaPresentacion) {

      texto +=
        ' - FECHA PRESENTACIÓN: ' +
        fechaPresentacion;

    }

    if (observacion) {

      texto +=
        ' - OBSERVACIÓN: ' +
        observacion;

    }

    return texto;

  }


  /* =====================================================
    26. REGISTRAR NOVEDAD
    ===================================================== */

  function registrarNovedad(token, datos) {

    const sesion =
      validarSesion_(token);

    if (!datos) {

      throw new Error(
        'No se recibieron datos de la novedad.'
      );

    }

    const cedula =
      normalizarCC_(
        datos.CC ||
        datos.cedula ||
        datos.CEDULA
      );

    if (!cedula) {

      throw new Error(
        'Debe indicar la cédula del funcionario.'
      );

    }

    const funcionario =
      obtenerFuncionarioPorCedula_(cedula);

    if (!funcionario) {

      throw new Error(
        'El funcionario no existe en LISTADO_BASE.'
      );

    }

    if (
      !validarDependencia_(
        sesion,
        funcionario.dependencia
      )
    ) {

      throw new Error(
        'No tiene autorización para registrar novedades en esta dependencia.'
      );

    }

    const novedad =
      String(
        datos.NOVEDAD ||
        datos.novedad ||
        ''
      ).trim();

    if (!novedad) {

      throw new Error(
        'Debe seleccionar una NOVEDAD.'
      );

    }

    const hoja =
      validarEstructuraNovedades_();

    const fila =
      new Array(
        COLUMNAS_NOVEDADES.length
      ).fill('');

    /*
    * A — GR
    */
    fila[0] =
      funcionario.grado || '';

    /*
    * B — APELLIDOS Y NOMBRES
    */
    fila[1] =
      funcionario.funcionario || '';

    /*
    * C — CC
    */
    fila[2] =
      funcionario.cedula || cedula;

    /*
    * D — DEPENDENCIA
    */
    fila[3] =
      funcionario.dependencia || '';

    /*
    * E — NOVEDAD
    */
    fila[4] =
      novedad;

    /*
    * F — DESCRIPCION
    */
    fila[5] =
      datos.DESCRIPCION ||
      datos.descripcion ||
      '';

    /*
    * G — Dias
    */
    fila[6] =
      datos.Dias ||
      datos.dias ||
      '';

    /*
    * H — Fecha INICIAL
    */
    fila[7] =
      convertirFecha_(datos['Fecha INICIAL']);

    /*
    * I — Fecha PRESENTACION
    */
    fila[8] =
      convertirFecha_(datos['Fecha PRESENTACION']);

    /*
    * J — Observacion
    */
    fila[9] =
      datos.Observacion ||
      datos.observacion ||
      '';

    /*
    * K — RV
    */
    fila[10] =
      datos.RV ||
      '';

    /*
    * L — NIV
    */
    fila[11] =
      funcionario.nivel || '';

    /*
    * M — Turno
    */
    fila[12] =
      funcionario.turno || '';

    /*
    * N — Placa_Chip
    */
    fila[13] =
      funcionario.placaChip || '';

    /*
    * O — TEXTO
    */
    fila[14] =
      datos.TEXTO ||
      generarTextoNovedad_({

        NOVEDAD: fila[4],

        DESCRIPCION: fila[5],

        Dias: fila[6],

        'Fecha INICIAL': fila[7],

        'Fecha PRESENTACION': fila[8],

        Observacion: fila[9]

      });

    hoja.appendRow(fila);

    const nuevaFila =
      hoja.getLastRow();

    /*
    * Formato de fechas
    */
    hoja
      .getRange(nuevaFila, 8)
      .setNumberFormat('dd/MM/yyyy');

    hoja
      .getRange(nuevaFila, 9)
      .setNumberFormat('dd/MM/yyyy');

    registrarAuditoria_(
      sesion.usuario,
      'REGISTRO_NOVEDAD',
      cedula,
      'Novedad: ' + novedad
    );

    return {

      estado: true,

      mensaje:
        'Novedad registrada correctamente.',

      fila: nuevaFila,

      novedad: fila

    };

  }


  /* =====================================================
    27. CONVERTIR FECHA
    ===================================================== */

  function convertirFecha_(valor) {

    if (
      valor === null ||
      valor === undefined ||
      valor === ''
    ) {

      return '';

    }

    if (
      Object.prototype.toString.call(valor) ===
      '[object Date]'
    ) {

      return valor;

    }

    const fecha =
      new Date(valor);

    if (
      isNaN(fecha.getTime())
    ) {

      return '';

    }

    return fecha;

  }

/* =====================================================
  28. CONSULTA POR TURNO
  ===================================================== */

function consultarPorTurno(token, turno) {

  const sesion = validarSesion_(token);

  turno = normalizarTexto_(turno);

  if (['A', 'B', 'C'].indexOf(turno) === -1) {
    throw new Error('El turno debe ser A, B o C.');
  }

  const ss = getSS_();

  const hojaBase =
    ss.getSheetByName(HOJA_BASE);

  if (!hojaBase) {
    throw new Error(
      'No existe la hoja LISTADO_BASE.'
    );
  }

  const valores =
    hojaBase
      .getDataRange()
      .getValues();

  if (valores.length < 2) {

    return {
      estado: true,
      turno: turno,
      datos: [],
      total: 0
    };

  }

  const encabezados =
    valores[0].map(function(x) {
      return String(x || '').trim();
    });

  const idxTurno =
    encabezados.findIndex(function(x) {
      return normalizarTexto_(x) === 'TURNO';
    });

  const idxCC =
    encabezados.findIndex(function(x) {
      return normalizarTexto_(x) === 'CEDULA';
    });

  if (idxTurno === -1) {
    throw new Error(
      'No se encontró la columna TURNO en LISTADO_BASE.'
    );
  }

  if (idxCC === -1) {
    throw new Error(
      'No se encontró la columna CEDULA en LISTADO_BASE.'
    );
  }

  const resultado = [];

  for (
    let i = 1;
    i < valores.length;
    i++
  ) {

    const fila = valores[i];

    const turnoFila =
      normalizarTexto_(
        fila[idxTurno]
      );

    if (turnoFila !== turno) {
      continue;
    }

    /*
     * Se construye el funcionario
     * directamente desde LISTADO_BASE.
     */

    const funcionario =
      construirFuncionario_(
        encabezados,
        fila,
        i + 1
      );

    const cedula =
      String(
        funcionario.cedula || ''
      ).trim();

    let historial = [];

    /*
     * Obtener historial de novedades
     * asociado a la cédula.
     */

    if (cedula) {

      try {

        historial =
          obtenerHistorialCC_(cedula) || [];

      } catch (e) {

        historial = [];

      }

    }

    funcionario.novedades =
      historial;

    funcionario.novedadesActivas =
      historial;

    resultado.push(
      funcionario
    );

  }

  /*
   * Auditoría
   */

  try {

    registrarAuditoria_(
      sesion.usuario,
      'CONSULTA_TURNO',
      '',
      'Turno ' +
        turno +
        ' — Funcionarios: ' +
        resultado.length
    );

  } catch (e) {

    console.log(
      'No fue posible registrar auditoría: ' +
      e.message
    );

  }

  return {

    estado: true,

    turno: turno,

    datos: resultado,

    total: resultado.length

  };

}

  function obtenerDashboard(token) {

    const sesion =
      validarSesion_(token);

    const ss =
      getSS_();

    const hojaBase =
      ss.getSheetByName(HOJA_BASE);

    if (!hojaBase) {

      throw new Error(
        'No existe LISTADO_BASE.'
      );

    }

    const valores =
      hojaBase
        .getDataRange()
        .getValues();

    if (valores.length < 2) {

      return {

        estado: true,

        total: 0,

        turnos: {
          A: 0,
          B: 0,
          C: 0
        },

        novedades: {}

      };

    }

    const encabezados =
      valores[0].map(function(x) {

        return String(x).trim();

      });

    const idxTurno =
      encabezados.indexOf('TURNO');

    const idxDep =
      encabezados.indexOf('DEPENDENCIA');

    const turnos = {
      A: 0,
      B: 0,
      C: 0
    };

    let total = 0;

    for (let i = 1; i < valores.length; i++) {

      const dependencia =
        idxDep !== -1
          ? valores[i][idxDep]
          : '';

      if (
        !validarDependencia_(
          sesion,
          dependencia
        )
      ) {

        continue;

      }

      total++;

      if (idxTurno !== -1) {

        const turno =
          normalizarTexto_(
            valores[i][idxTurno]
          );

        if (
          Object.prototype.hasOwnProperty.call(
            turnos,
            turno
          )
        ) {

          turnos[turno]++;

        }

      }

    }

    /*
    * Estadísticas de novedades
    */

    const hojaNov =
      ss.getSheetByName(
        HOJA_NOVEDADES
      );

    const novedades = {};

    if (
      hojaNov &&
      hojaNov.getLastRow() >= 2
    ) {

      const datosNov =
        hojaNov
          .getRange(
            2,
            1,
            hojaNov.getLastRow() - 1,
            COLUMNAS_NOVEDADES.length
          )
          .getValues();

      const idxDepNov = 3;
      const idxNovedad = 4;
      const idxTurnoNov = 12;

      datosNov.forEach(function(fila) {

        const dependencia =
          fila[idxDepNov];

        if (
          !validarDependencia_(
            sesion,
            dependencia
          )
        ) {

          return;

        }

        const nombre =
          normalizarTexto_(
            fila[idxNovedad]
          );

        if (!nombre) {
          return;
        }

        if (!novedades[nombre]) {

          novedades[nombre] = {

            total: 0,

            A: 0,

            B: 0,

            C: 0

          };

        }

        novedades[nombre].total++;

        const turno =
          normalizarTexto_(
            fila[idxTurnoNov]
          );

        if (
          novedades[nombre][turno] !==
          undefined
        ) {

          novedades[nombre][turno]++;

        }

      });

    }

    registrarAuditoria_(
      sesion.usuario,
      'CONSULTA_DASHBOARD',
      '',
      'Dashboard consultado'
    );

    return {

      estado: true,

      total: total,

      turnos: turnos,

      novedades: novedades

    };

  }

  /* =====================================================
    31. ANULAR NOVEDAD
    =====================================================

    IMPORTANTE:

    La hoja NOVEDADES NO tiene columna ESTADO.

    Por ello NO se modifica ni elimina físicamente
    el registro.

    Esta función registra la operación en AUDITORIA.

    Si posteriormente deseas una anulación visual,
    podemos incorporar una hoja HISTORIAL_ANULACIONES
    sin modificar NOVEDADES.
    ===================================================== */

  function anularNovedad(token, fila, motivo) {

    const sesion =
      validarSesion_(token);

    if (
      normalizarTexto_(sesion.rol) !==
      'ADMINISTRADOR'
    ) {

      throw new Error(
        'Solo un ADMINISTRADOR puede realizar esta operación.'
      );

    }

    fila =
      Number(fila);

    if (!fila || fila < 2) {

      throw new Error(
        'Fila de novedad inválida.'
      );

    }

    const hoja =
      validarEstructuraNovedades_();

    if (
      fila > hoja.getLastRow()
    ) {

      throw new Error(
        'La fila indicada no existe.'
      );

    }

    const registro =
      hoja
        .getRange(
          fila,
          1,
          1,
          COLUMNAS_NOVEDADES.length
        )
        .getValues()[0];

    const cc =
      registro[2];

    const novedad =
      registro[4];

    /*
    * No se elimina el registro.
    */

    registrarAuditoria_(
      sesion.usuario,
      'ANULACION_NOVEDAD',
      cc,
      'Fila: ' +
        fila +
        ' | Novedad: ' +
        novedad +
        ' | Motivo: ' +
        (motivo || '')
    );

    return {

      estado: true,

      mensaje:
        'La operación fue registrada en AUDITORIA. ' +
        'El registro original de NOVEDADES se conserva.'

    };

  }

      validarSesion_(token);

        .getDataRange()
        .getValues();

        i + 1;

  function crearUsuario(token, datos) {

    const sesion =
      validarSesion_(token);

    if (
      normalizarTexto_(sesion.rol) !==
      'ADMINISTRADOR'
    ) {

      throw new Error(
        'Solo un ADMINISTRADOR puede crear usuarios.'
      );

    }

    datos =
      datos || {};

    const usuario =
      normalizarTexto_(
        datos.usuario
      );

    const clave =
      String(
        datos.clave || ''
      );

    const rol =
      normalizarTexto_(
        datos.rol ||
        'OPERADOR'
      );

    const dependencia =
      normalizarTexto_(
        datos.dependencia || ''
      );

    if (!usuario) {

      throw new Error(
        'Debe indicar el usuario.'
      );

    }

    if (clave.length < 6) {

      throw new Error(
        'La contraseña debe tener mínimo 6 caracteres.'
      );

    }

    if (
      ['ADMINISTRADOR', 'OPERADOR']
        .indexOf(rol) === -1
    ) {

      throw new Error(
        'Rol no válido.'
      );

    }

    const ss =
      getSS_();

    let hoja =
      ss.getSheetByName(
        HOJA_USUARIOS
      );

    if (!hoja) {

      hoja =
        ss.insertSheet(
          HOJA_USUARIOS
        );

      hoja
        .getRange(1, 1, 1, 7)
        .setValues([[
          'USUARIO',
          'PASSWORD_HASH',
          'ROL',
          'DEPENDENCIA',
          'ESTADO',
          'FECHA_CREACION',
          'ULTIMO_ACCESO'
        ]]);

    }

    const existente =
      obtenerUsuario_(usuario);

    if (existente) {

      throw new Error(
        'El usuario ya existe.'
      );

    }

    hoja.appendRow([

      usuario,

      hashPassword_(clave),

      rol,

      dependencia,

      'ACTIVO',

      new Date(),

      ''

    ]);

    registrarAuditoria_(
      sesion.usuario,
      'CREAR_USUARIO',
      '',
      'Usuario creado: ' +
        usuario
    );

    return {

      estado: true,

      mensaje:
        'Usuario creado correctamente.'

    };

  }

  function cambiarPassword(
    token,
    usuarioDestino,
    nuevaClave
  ) {

    const sesion =
      validarSesion_(token);

    usuarioDestino =
      normalizarTexto_(

    nuevaClave =
      String(
        nuevaClave || ''
      );

    const esAdmin =
      normalizarTexto_(
        sesion.rol
      ) === 'ADMINISTRADOR';

    const esPropioUsuario =
      normalizarTexto_(
        sesion.usuario
      ) === usuarioDestino;

    if (
      !esAdmin &&
      !esPropioUsuario
    ) {

      throw new Error(
        'No tiene autorización para cambiar esta contraseña.'
      );

    }

    if (nuevaClave.length < 6) {

      throw new Error(
        'La nueva contraseña debe tener mínimo 6 caracteres.'
      );

    }

    const usuario =
      obtenerUsuario_(
        usuarioDestino
      );

    if (!usuario) {

      throw new Error(
        'El usuario no existe.'
      );

    }

    const ss =
      getSS_();

    const hoja =
      ss.getSheetByName(
        HOJA_USUARIOS
      );

    const encabezados =
      obtenerEncabezados_(hoja);

    const idxHash =
      encabezados.indexOf(
        'PASSWORD_HASH'
      );

    hoja
      .getRange(
        usuario.fila,
        idxHash + 1
      )
      .setValue(
        hashPassword_(nuevaClave)
      );

    registrarAuditoria_(
      sesion.usuario,
      'CAMBIO_PASSWORD',
      '',
      'Usuario afectado: ' +
        usuarioDestino
    );

    return {

      estado: true,

      mensaje:
        'Contraseña actualizada correctamente.'

    };

  }

  function obtenerConsecutivoReporte_() {

    const lock =
      LockService.getScriptLock();

    lock.waitLock(30000);

    try {

      const ss =
        getSS_();

      let hoja =
        ss.getSheetByName(
          HOJA_CONFIG
        );

      if (!hoja) {

        hoja =
          ss.insertSheet(
            HOJA_CONFIG
          );

      }

      let valor =
        hoja
          .getRange('B1')
          .getValue();

      if (
        !valor ||
        isNaN(Number(valor))
      ) {

        valor = 0;

      }

      valor =
        Number(valor) + 1;

      hoja
        .getRange('A1')
        .setValue(
          'CONSECUTIVO_REPORTE'
        );

      hoja
        .getRange('B1')
        .setValue(
          valor
        );

      return (
        'NVD-' +
        Utilities.formatDate(
          new Date(),
          Session.getScriptTimeZone(),
          'yyyy'
        ) +
        '-' +
        ('000000' + valor)
          .slice(-6)
      );

    } finally {

      lock.releaseLock();

    }

  }

  function generarPDFInstitucional(
    token,
    turno
  ) {

    const sesion =
      validarSesion_(token);

    turno =
      normalizarTexto_(turno);

    if (
      ['A', 'B', 'C']
        .indexOf(turno) === -1
    ) {

      throw new Error(
        'Turno inválido.'
      );

    }

    const reporte =
      consultarPorTurno(
        token,
        turno
      );

    const consecutivo =
      obtenerConsecutivoReporte_();

    const fecha =
      new Date();

    /*
    * Preparar datos para ReportePDF.html
    */

    const plantilla =
      HtmlService
        .createTemplateFromFile(
          'ReportePDF'
        );

    plantilla.turno =
      turno;

    plantilla.fecha =
      formatearFecha_(
        fecha
      );

    plantilla.consecutivo =
      consecutivo;

    plantilla.usuario =
      sesion.usuario;

    plantilla.funcionarios =
      reporte.datos;

    /*
    * Generar HTML
    */

    const html =
      plantilla
        .evaluate()
        .getContent();

    const blob =
      Utilities
        .newBlob(
          html,
          MimeType.HTML,
          consecutivo + '.html'
        )
        .getAs(
          MimeType.PDF
        );

    blob.setName(
      consecutivo +
      '_TURNO_' +
      turno +
      '.pdf'
    );

    /*
    * Carpeta de reportes
    */

    const carpeta =
      obtenerCarpetaReportes_();

    const archivo =
      carpeta.createFile(
        blob
      );

    /*
    * Registrar reporte
    */

    registrarReporte_(
      consecutivo,
      fecha,
      sesion.usuario,
      turno,
      archivo
    );

    registrarAuditoria_(
      sesion.usuario,
      'GENERAR_PDF',
      '',
      consecutivo +
        ' | Turno ' +
        turno
    );

    return {

      estado: true,

      consecutivo:
        consecutivo,

      nombre:
        archivo.getName(),

      url:
        archivo.getUrl()

    };

  }

          .getFolderById(
            String(idCarpeta).trim()
          );

      } catch (e) {

        throw new Error(
          'El ID de carpeta indicado en CONFIG!B2 no es válido.'
        );

      }

    }

    /*
    * Si no existe ID, crear carpeta
    * dentro de la carpeta del Spreadsheet.
    */

    const archivo =
      DriveApp.getFileById(
        ss.getId()
      );

    const padres =
      archivo.getParents();

    let carpetaPadre;

    if (padres.hasNext()) {

      carpetaPadre =
        padres.next();

    } else {

      carpetaPadre =
        DriveApp.getRootFolder();

    }

    const nombre =
      'SISTEMA_GESTION_NOVEDADES';

    const existentes =
      carpetaPadre
        .getFoldersByName(
          nombre
        );

    if (existentes.hasNext()) {

      return existentes.next();

    }

    return carpetaPadre
      .createFolder(
        nombre
      );

  }

    ===================================================== */

  function obtenerReportes(token) {

    const sesion =
      validarSesion_(token);

    const ss =
      getSS_();

    const hoja =
      ss.getSheetByName(
        HOJA_REPORTES
      );

    if (!hoja) {
      return [];
    }

    const datos =
      hoja.getDataRange()
        .getValues();

    if (datos.length < 2) {
      return [];
    }

    /*
    * Todos los reportes son visibles
    * para administradores.
    */

    if (
      normalizarTexto_(sesion.rol) ===
      'ADMINISTRADOR'
    ) {

      return datos
        .slice(1)
        .map(function(fila) {

          return {

            consecutivo: fila[0],

            fecha: fila[1],

            usuario: fila[2],

            turno: fila[3],

            archivo: fila[4],

            url: fila[5]

          };

        });

    }

    /*
    * Operadores únicamente ven
    * sus propios reportes.
    */

    return datos
      .slice(1)
      .filter(function(fila) {

        return (
          normalizarTexto_(fila[2]) ===
          normalizarTexto_(sesion.usuario)
        );

      })
      .map(function(fila) {

        return {

          consecutivo: fila[0],

          fecha: fila[1],

          usuario: fila[2],

          turno: fila[3],

          archivo: fila[4],

          url: fila[5]

        };

      });

  }

          'ESTADO',
          'FECHA_CREACION',
          'ULTIMO_ACCESO'
        ]]);

    }

    */

    let reportes =
      ss.getSheetByName(
        HOJA_REPORTES
      );

    if (!reportes) {

      reportes =
        ss.insertSheet(
          HOJA_REPORTES
        );

      reportes
        .getRange(1, 1, 1, 6)
        .setValues([[
          'CONSECUTIVO',
          'FECHA',
          'USUARIO',
          'TURNO',
          'ARCHIVO',
          'URL'
        ]]);

    }

    /*
    * CONFIG
    */

    let config =
      ss.getSheetByName(
        HOJA_CONFIG
      );

    if (!config) {

      config =
        ss.insertSheet(
          HOJA_CONFIG
        );

    }

    config
      .getRange('A1')
      .setValue(
        'CONSECUTIVO_REPORTE'
      );

    if (
      !config
        .getRange('B1')
        .getValue()
    ) {

      config
        .getRange('B1')
        .setValue(0);

    }

    config
      .getRange('A2')
      .setValue(
        'ID_CARPETA_REPORTES'
      );

    return {

      estado: true,

      mensaje:
        'Estructura del sistema inicializada correctamente.'

    };

  }

  /**
   * Genera un hash SHA-256 en formato hexadecimal.
   */
  function sha256_(texto) {

    texto = String(texto || '');

    const bytes = Utilities.computeDigest(
      Utilities.DigestAlgorithm.SHA_256,
      texto,
      Utilities.Charset.UTF_8
    );

    return bytes
      .map(function(byte) {
        const valor = byte < 0 ? byte + 256 : byte;
        return ('0' + valor.toString(16)).slice(-2);
      })
      .join('');
  }

function diagnosticarTurnos() {

  const ss = getSS_();

  const hoja = ss.getSheetByName(HOJA_BASE);

  if (!hoja) {
    throw new Error('No existe LISTADO_BASE.');
  }

  const datos =
    hoja.getDataRange().getValues();

  const encabezados =
    datos[0].map(function(x) {
      return String(x || '').trim();
    });

  const idxTurno =
    encabezados.findIndex(function(x) {
      return normalizarTexto_(x) === 'TURNO';
    });

  if (idxTurno === -1) {
    throw new Error(
      'No se encontró la columna TURNO.'
    );
  }

  const conteo = {};

  for (let i = 1; i < datos.length; i++) {

    const valor =
      String(
        datos[i][idxTurno] || ''
      ).trim();

    if (!conteo[valor]) {
      conteo[valor] = 0;
    }

    conteo[valor]++;
  }

  Logger.log(
    JSON.stringify(
      conteo,
      null,
      2
    )
  );

  return conteo;
}
